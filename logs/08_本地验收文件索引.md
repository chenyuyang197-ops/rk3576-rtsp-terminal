### NV12 → PNG（无 ffmpeg 时）

```bash
python3 -c "
import numpy as np
from pathlib import Path
from PIL import Image
w,h=720,1280
p='/tmp/rk3576-ch0.yuv'
yuv=np.frombuffer(Path(p).read_bytes()[:w*h*3//2],dtype=np.uint8)
y=yuv[:w*h].reshape(h,w); uv=yuv[w*h:].reshape(h//2,w)
u=uv[:,0::2].repeat(2,0).repeat(2,1)[:h,:w]
v=uv[:,1::2].repeat(2,0).repeat(2,1)[:h,:w]
c=y.astype(float)-16; d=u.astype(float)-128; e=v.astype(float)-128
r=np.clip(1.164*c+1.596*e,0,255).astype(np.uint8)
g=np.clip(1.164*c-0.392*d-0.813*e,0,255).astype(np.uint8)
b=np.clip(1.164*c+2.017*d,0,255).astype(np.uint8)
Image.fromarray(np.stack([r,g,b],-1)).save('/tmp/rk3576-ch0.png')
print('ok')
"
```
